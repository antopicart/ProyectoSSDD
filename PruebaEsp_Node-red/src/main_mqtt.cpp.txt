// ======================================================
// LIBRERÍAS
// ======================================================

// Incluye las funciones principales de Arduino.
// Permite usar setup(), loop(), delay(), Serial, etc.
#include <Arduino.h>

// Incluye la librería necesaria para manejar el Wi-Fi del ESP8266.
#include <ESP8266WiFi.h>

// Incluye la librería PubSubClient.
// Esta librería permite que el ESP8266 se comunique utilizando MQTT.
#include <PubSubClient.h>


// ======================================================
// CONFIGURACIÓN DEL WI-FI
// ======================================================

// Guarda el nombre de la red Wi-Fi a la que se conectará el ESP8266.
const char* nombreWifi = "VideoDigital-Neto";

// Guarda la contraseña de la red Wi-Fi.
const char* contrasenaWifi = "neto0290";


// ======================================================
// CONFIGURACIÓN DE MQTT
// ======================================================

// Dirección IP de la computadora donde está funcionando Mosquitto.
// Esta es la IP de nuestra PC, NO la IP del ESP8266.
const char* servidorMqtt = "192.168.101.8";

// Puerto utilizado por Mosquitto para MQTT.
// El puerto estándar MQTT sin cifrado es 1883.
const int puertoMqtt = 1883;


// ======================================================
// CREACIÓN DE LOS CLIENTES
// ======================================================

// Crea un cliente de red.
// Este cliente utiliza la conexión Wi-Fi del ESP8266.
WiFiClient clienteWifi;

// Crea el cliente MQTT utilizando el cliente Wi-Fi anterior.
// Con este objeto podremos conectarnos a Mosquitto
// y publicar mensajes MQTT.
PubSubClient clienteMqtt(clienteWifi);


// ======================================================
// SETUP
// ======================================================

// setup() se ejecuta una sola vez cuando el ESP8266
// se enciende o se reinicia.
void setup()
{
    // Inicia la comunicación serie entre el ESP8266 y la computadora.
    // El Monitor Serie también debe estar configurado a 115200.
    Serial.begin(115200);

    // Imprime una línea vacía.
    Serial.println();


    // ==================================================
    // CONEXIÓN AL WI-FI
    // ==================================================

    // Informa que comenzará el intento de conexión.
    Serial.println("Conectando al Wi-Fi...");

    // Inicia la conexión utilizando el nombre
    // y la contraseña definidos anteriormente.
    WiFi.begin(nombreWifi, contrasenaWifi);

    // Mientras el ESP8266 no esté conectado al Wi-Fi,
    // permanece dentro de este while.
    while (WiFi.status() != WL_CONNECTED)
    {
        // Espera medio segundo.
        delay(500);

        // Muestra un punto para indicar
        // que todavía está intentando conectarse.
        Serial.print(".");
    }

    // Si salió del while significa que logró conectarse.
    Serial.println();

    // Informa que la conexión Wi-Fi fue exitosa.
    Serial.println("Wi-Fi conectado");

    // Muestra la dirección IP que recibió el ESP8266.
    Serial.print("Direccion IP: ");
    Serial.println(WiFi.localIP());


    // ==================================================
    // CONFIGURACIÓN DEL BROKER MQTT
    // ==================================================

    // Le indica al cliente MQTT dónde está Mosquitto.
    // servidorMqtt = IP de la PC.
    // puertoMqtt = 1883.
    clienteMqtt.setServer(servidorMqtt, puertoMqtt);


    // ==================================================
    // CONEXIÓN CON MOSQUITTO
    // ==================================================

    // Informa que comenzará el intento de conexión MQTT.
    Serial.println("Conectando al broker MQTT...");

    // Intenta conectar el ESP8266 con Mosquitto.
    // "ESP8266_cliente" es el nombre con el que
    // nuestro dispositivo se identifica ante el broker.
    if (clienteMqtt.connect("ESP8266_cliente"))
    {
        // Si connect() devuelve true,
        // significa que la conexión fue exitosa.
        Serial.println("Conectado a Mosquitto");


        // ==============================================
        // PUBLICACIÓN DEL MENSAJE MQTT
        // ==============================================

        // Publica el mensaje "hola" en el tópico "prueba".
        //
        // "prueba" = tópico
        // "hola"   = mensaje
        clienteMqtt.publish("prueba", "hola");

        // Muestra en el Monitor Serie
        // que el mensaje fue enviado.
        Serial.println("Mensaje MQTT enviado: hola");
    }
    else
    {
        // Si connect() devuelve false,
        // significa que no pudo conectarse con Mosquitto.
        Serial.println("Error al conectar con Mosquitto");

        // Muestra el código de error MQTT.
        Serial.print("Codigo de error MQTT: ");
        Serial.println(clienteMqtt.state());
    }
}


// ======================================================
// LOOP
// ======================================================

// loop() se ejecuta continuamente después de setup().
void loop()
{
    // Por ahora queda vacío.
    //
    // Más adelante utilizaremos esta parte para
    // mantener la conexión MQTT y enviar los datos
    // de los sensores.
}
